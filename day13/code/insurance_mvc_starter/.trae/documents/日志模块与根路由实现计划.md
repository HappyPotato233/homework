## 四、详细设计

### 4.1 修改 `app/models/operation_log.py`

追加 `paginate` 类方法，对齐 `Customer.paginate` 风格：

```python
@classmethod
def paginate(cls, db, page, per_page, user_id=None, action=None):
    """分页查询操作日志，返回 {items, total, page, per_page, pages}"""
    query = db.query(cls)
    if user_id:
        query = query.filter(cls.user_id == user_id)
    if action:
        query = query.filter(cls.action == action)
    total = query.count()
    items = query.order_by(cls.id.desc()).offset((page-1)*per_page).limit(per_page).all()
    pages = (total + per_page - 1) // per_page if per_page else 0
    return {"items": [_to_dict(log) for log in items], "total": total,
            "page": page, "per_page": per_page, "pages": pages}
```

追加模块级 `_to_dict` 函数，返回 `{id, user_id, action, details, created_at(ISO8601)}`。

### 4.2 新建 `app/services/log_service.py`

def query_logs(db, page, per_page, user_id=None, action=None):
    """调用 OperationLog.paginate，返回分页结果"""

### 4.3 新建 `app/api/v1/log.py`

```python
bp = Blueprint("log", __name__)

VALID_ACTIONS = {"model_training", "prediction", "model_import",
                  "email_generation", "email_update", "email_mark", "email_delete"}

@bp.route("", methods=["GET"])
@role_required("admin")
def query():
    page = request.args.get("page", 1, type=int)
    per_page = request.args.get("per_page", 50, type=int)
    # 校验 page/per_page > 0
    if page < 1 or per_page < 1:
        raise BizException(1001, "参数校验错误，page和per_page必须大于0", 400)
    user_id = request.args.get("user_id", type=int)
    action = request.args.get("action")
    if action and action not in VALID_ACTIONS:
        raise BizException(1001, f"非法action值: {action}", 400)
    db = get_db()
    data = log_service.query_logs(db, page, per_page, user_id, action)
    return json(data)
```

蓝图 URL 前缀：`/api/v1/logs`，路由 `""` -> 完整路径 `/api/v1/logs`。

### 4.4 新建 `app/utils/file_helper.py`

```python
def read_static_file(filename: str) -> str:
    """读取 app/static/ 下的文件内容，文件缺失抛 BizException(5000)"""
    path = BASE_DIR / "app" / "static" / filename
    if not path.exists():
        raise BizException(5000, "前端页面资源缺失", 500)
    return path.read_text(encoding="utf-8")
```
